# ================= LOAD KEYWORD =================
def load_keywords_with_desc():
    keyword_docs = []

    for name, func in inspect.getmembers(KWCommon, predicate=inspect.isfunction):
        if name.startswith("_"):
            continue

        doc = inspect.getdoc(func) or ""
        keyword_docs.append(f"{name}: {doc}")

    return keyword_docs


# ================= PARSE JSON =================
def extract_json(raw: str):
    raw = raw.strip()

    raw = re.sub(r"```json", "", raw)
    raw = re.sub(r"```", "", raw)
    raw = re.sub(r"//.*", "", raw)
    raw = re.sub(r"/\*.*?\*/", "", raw, flags=re.DOTALL)
    raw = re.sub(r",\s*}", "}", raw)
    raw = re.sub(r",\s*]", "]", raw)

    match = re.search(r"\[.*\]", raw, re.DOTALL)

    if match:
        return json.loads(match.group(0))

    raise Exception("Không parse được JSON từ AI")


# ================= SETUP =================
def inject_setup_teardown(result):
    config = load_config()
    base_url = config.get("base_url", "")

    for tc in result:
        steps = tc.get("steps", [])

        keywords = [s.get("keyword", "") for s in steps]

        if "OPEN_URL" not in keywords:
            steps.insert(0, {
                "keyword": "OPEN_URL",
                "locator": "",
                "value": base_url
            })

        if "CLOSE_BROWSER" not in keywords:
            steps.append({
                "keyword": "CLOSE_BROWSER",
                "locator": "",
                "value": ""
            })

    return result


# ================= KEYWORD =================
def get_all_valid_keywords():
    return set([
        name
        for name, _ in inspect.getmembers(KWCommon, predicate=inspect.isfunction)
        if not name.startswith("_")
    ])


def auto_fix_keyword(step_text):
    return extract_action(step_text)


# ================= NORMALIZE TEXT =================
def normalize_text(text):
    if not text:
        return ""

    return (
        text
        .replace("Hệ thống hiển thị", "")
        .replace("thông báo", "")
        .replace("xuất hiện", "")
        .replace('"', "")
        .strip()
    )


# ================= VALIDATE =================
def validate_and_fix(result, testcases, locator_context):

    for tc, original_tc in zip(result, testcases):

        ai_steps = tc.get("steps", [])
        raw_steps = original_tc.get("steps", [])
        expected_results = original_tc.get("expected_result", [])

        clean_steps = []
        seen_verify = set()

        for idx, step_obj in enumerate(ai_steps):

            keyword = step_obj.get("keyword", "")

            if keyword in ["OPEN_URL", "CLOSE_BROWSER"]:
                clean_steps.append(step_obj)
                continue

            if idx >= len(raw_steps):
                break

            raw_step = raw_steps[idx]

            # ================= KEYWORD (SAFE) =================
            old_kw = step_obj.get("keyword", "")

            if old_kw in ["", None, "UNKNOWN"]:
                new_kw = auto_fix_keyword(raw_step)
            else:
                new_kw = old_kw

            step_obj["keyword"] = new_kw

            # ================= LOCATOR CONTEXT =================
            locator_text = raw_step

            if new_kw.startswith("VERIFY"):
                expected_text = " ".join(expected_results)
                locator_text = f"[VERIFY] {raw_step} [EXPECTED] {expected_text}"

            elif new_kw.startswith("INPUT"):
                locator_text = f"[INPUT] {raw_step}"

            elif new_kw.startswith("CLICK"):
                locator_text = f"[CLICK] {raw_step}"

            # ================= LOCATOR FIX (SAFE ONLY) =================
            old_locator = step_obj.get("locator", "")

            if not old_locator:
                new_locator = map_locator_semantic(locator_text, locator_context)
                step_obj["locator"] = new_locator

            # ================= VALUE FIX =================
            if new_kw in ["VERIFY_ELEMENT_TEXT_EQUALS", "VERIFY_TEXT_CONTAINS"]:

                if expected_results:

                    expected_text = normalize_text(expected_results[0])
                    step_obj["value"] = expected_text

            # ================= REMOVE DUPLICATE VERIFY =================
            if new_kw in ["VERIFY_ELEMENT_TEXT_EQUALS", "VERIFY_TEXT_CONTAINS"]:

                current_key = (
                    new_kw,
                    step_obj.get("locator"),
                    str(step_obj.get("value", "")).strip()
                )

                if current_key in seen_verify:
                    continue

                seen_verify.add(current_key)

            clean_steps.append(step_obj)

        tc["steps"] = clean_steps

    return result


# ================= MAIN =================
def generate_keyword_steps(testcases, locator_path, topic):

    keyword_docs = load_keywords_with_desc()
    keyword_str = "\n".join(keyword_docs)

    with open(locator_path, "r", encoding="utf-8") as f:
        locators = yaml.safe_load(f)

    locator_context = []

    for key, value in locators.items():
        locator_context.append({
            "name": key,
            "type": normalize_locator_name(key),
            "semantic": value.get("semantic", ""),
            "by": value.get("by", "")
        })

    prompt = PROMPT.format(
        keywords=keyword_str,
        testcases=json.dumps(testcases, indent=2, ensure_ascii=False),
        locators=json.dumps(locator_context, indent=2, ensure_ascii=False),
        topic=topic
    )

    raw = call_llm(prompt)

    print("===== RAW AI OUTPUT =====")
    print(raw)

    result = extract_json(raw)
    result = inject_setup_teardown(result)
    result = validate_and_fix(result, testcases, locator_context)

    return result
